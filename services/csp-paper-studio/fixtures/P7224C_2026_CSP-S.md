#include <bits/stdc++.h>
using namespace std;

const int N = 300005;
string s;
int pref[N][26];
long long t[27];

int distinctCnt(int l, int r) {
    int ans = 0;
    for (int c = 0; c < 26; c++)
        if (pref[r][c] - pref[l - 1][c] > 0) ans++;
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> s;
    int n = (int)s.size();
    s = " " + s;

    for (int i = 1; i <= n; i++) {
        for (int c = 0; c < 26; c++)
            pref[i][c] = pref[i - 1][c];
        pref[i][s[i] - 'a']++;
    }

    for (int d = 1; d <= 26; d++) {
        int r1 = 0, r2 = 0;
        int c1 = 0, c2 = 0;
        int f1[26] = {}, f2[26] = {};

        for (int l = 1; l <= n; l++) {
            r1 = max(r1, l - 1);
            r2 = max(r2, l - 1);

            while (r1 + 1 <= n && c1 <= d) {
                int x = s[r1 + 1] - 'a';
                if (f1[x] == 0 && c1 == d) break;
                if (f1[x]++ == 0) c1++;
                r1++;
            }

            while (r2 + 1 <= n && c2 < d) {
                int x = s[r2 + 1] - 'a';
                if (f2[x]++ == 0) c2++;
                r2++;
            }

            if (c1 == d && c2 == d)
                t[d] += r1 - r2 + 1;

            int x = s[l] - 'a';
            if (--f1[x] == 0) c1--;
            if (--f2[x] == 0) c2--;
        }
    }

    cout << distinctCnt(1, n) << '\n';
    for (int d = 1; d <= 26; d++)
        if (t[d]) cout << t[d] << '\n';
    return 0;
}

 - （两组数的严格配对计数）给定两个整数数组 `a[1..n]`、`b[1..m]`，分别从两个数组中各选择 `k` 个元素。把选出的两组数分别从小到大排序，记为

```text
x1 ≤ x2 ≤ ... ≤ xk
y1 ≤ y2 ≤ ... ≤ yk
```

数组中的元素按原下标区分：即使两个元素数值相同，只要选择的下标不同，就视为不同的选择方案。

求合法选择方案数，对 `1000000009` 取模。数据范围：`1≤n,m≤1000`，`1≤k≤10`。

下面程序使用动态规划。`f[cur][j][t]` 表示在当前处理到某个 `a[i]` 时，与 `b[1..j]` 相关、已经选出 `t` 对数的方案数；第一维采用滚动数组。

补全代码：

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 1010;
const int K = 12;
const long long MOD = 1000000009LL;

int n, m, k;
int a[N], b[N];
long long f[2][N][K];

int main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= m; i++) cin >> b[i];

    sort(a + 1, a + n + 1);
    sort(b + 1, b + m + 1);

    for (int p = 0; p < 2; p++)
        for (int j = 0; j <= m; j++)
            f[p][j][0] = 1;

    for (int i = 1; i <= n; i++) {
        int cur = i & 1;
        int pre = cur ^ 1;
        for (int j = 1; j <= m; j++) {
            for (int t = 1; t <= k; t++) {
                ①;
                ②;
                f[cur][j][t] = (f[cur][j][t] + f[cur][j - 1][t]) % MOD;
                ③;
                if (a[i] > b[j])
                    ④;
            }
        }
    }

    cout << ⑤ << '\n';
    return 0;
}
```

补全代码：

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 405;
const int M = N * N;

int n, m;
int fa[M], sz[M], cnt[M];
char a[N][N];

inline int id(int x, int y) {
    return ①;
}

int findSet(int x) {
    if (fa[x] != x)
        fa[x] = ②;
    return fa[x];
}

void mergeSet(int x, int y) {
    x = findSet(x);
    y = findSet(y);
    if (x == y) return;
    fa[x] = y;
    sz[y] += sz[x];
    cnt[y] += cnt[x];
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
        cin >> (a[i] + 1);

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (③) {
                a[i][j] = ④;
                cnt[id(i, j)]++;
            }
        }
    }

    for (int i = 1; i <= n * m; i++) {
        fa[i] = i;
        sz[i] = 1;
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (i < n && a[i][j] == a[i + 1][j])
                mergeSet(id(i, j), id(i + 1, j));
            if (j < m && a[i][j] == a[i][j + 1])
                mergeSet(id(i, j), id(i, j + 1));
        }
    }

    long long ans = 0;
    for (int i = 1; i <= n * m; i++) {
        if (findSet(i) == i)
            ans += ⑤;
    }

    cout << ans << '\n';
    return 0;
}
```
